cmake_minimum_required(VERSION 3.20)
project(onion_libtooling LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_VERSION_MAJOR EQUAL 18)
  message(FATAL_ERROR "OnionTrace extraction requires LLVM/Clang 18")
endif()
find_package(Clang REQUIRED CONFIG)
add_executable(onion-extract extract.cpp)
target_compile_features(onion-extract PRIVATE cxx_std_17)
target_include_directories(onion-extract SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
target_link_libraries(onion-extract PRIVATE clang-cpp LLVM)
