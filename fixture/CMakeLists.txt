cmake_minimum_required(VERSION 3.16)
project(onion_fixture LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
add_executable(onion_fixture fixture.c)
target_link_libraries(onion_fixture PRIVATE Threads::Threads)
target_compile_options(onion_fixture PRIVATE -Wall -Wextra -Werror)
if(DEFINED ONION_BUILD_ID)
  target_compile_definitions(onion_fixture PRIVATE BUILD_ID="${ONION_BUILD_ID}")
endif()
enable_testing()
add_test(NAME fixture_smoke COMMAND onion_fixture --run-id ctest-smoke --plan-id anchors --jobs 12)
add_test(NAME fixture_queue_wrap COMMAND onion_fixture --run-id ctest-wrap --plan-id anchors --jobs 1024)
add_test(NAME fixture_full_probes COMMAND onion_fixture --run-id ctest-full --plan-id all
  --jobs 1024 --seed 18446744073709551615
  --sites work.begin,work.end,branch.slow,branch.fast,retry.ok,retry.fail)
set_tests_properties(fixture_smoke fixture_queue_wrap fixture_full_probes PROPERTIES TIMEOUT 20)
