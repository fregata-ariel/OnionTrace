cmake_minimum_required(VERSION 3.20)
project(onion_build_inputs C)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VARIANT 1 CACHE STRING "Condition change fixture")
configure_file(config.h.in generated/config.h @ONLY)
add_library(shared STATIC shared.c)
add_library(object_worker OBJECT worker.c)
add_library(variant_a OBJECT variant.c)
add_library(variant_b OBJECT variant.c)
target_compile_definitions(variant_a PRIVATE MODE=1)
target_compile_definitions(variant_b PRIVATE MODE=2)
add_executable(nr-softmodem main.c)
add_executable(nr-uesoftmodem main.c)
target_link_libraries(nr-softmodem PRIVATE shared object_worker variant_a)
target_link_libraries(nr-uesoftmodem PRIVATE shared variant_b)
foreach(target shared object_worker variant_a variant_b nr-softmodem nr-uesoftmodem)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
endforeach()
add_library(unrelated STATIC unrelated.c)
